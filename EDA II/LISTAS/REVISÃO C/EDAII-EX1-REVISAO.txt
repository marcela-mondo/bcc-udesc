/*Dado um vetor de números inteiros v de tamanho n e um número k, retorne verdadeiro
se a soma de qualquer par de números em v for igual a k.
○ Exemplo: dado v = [10,15,3,7] e k = 17, a saída deve ser true, pois 10 + 7 é 17*/

#include <stdio.h>
#include <stdlib.h>

int main()
{
    int v=0, k, n, encontrou = 0;;
    int *vetor;
	printf("Informe o tamanho do vetor: ");
	scanf("%d", &n);
	
	vetor = malloc(n * sizeof(*vetor));
	if (vetor == NULL){
	    printf("Não foi possível alocar");
	}
	
	for (v=0; v<n; v++){
	    printf("\nInsira: ");
	    scanf("%d", &vetor[v]);
	}
		
	printf("\nInforme o valor de k: ");
	scanf("%d", &k);
	
	for (v = 0; v < n; v++){
	    for (int i = v + 1; i < n; i++){
	        if (vetor[v] + vetor[i] == k){
	            encontrou = 1;
	        }
	    }
	}

	if (encontrou)
	    printf("\ntrue");
	else
	    printf("\nfalse");

	free(vetor);

	
	return 0;
}
